    77→        if (themeSetting === 'light') { theme = 'light'; } else if (themeSetting === 'dark') { theme = 'dark'; } else if (themeSetting) {
    78→          try {
    79→            var parsed = JSON.parse(themeSetting);
    80→            if (parsed.accent === 'purple') { accent = '#855cd6'; } else if (parsed.accent === 'blue') { accent = '#4c97ff'; }

    82→          } catch (e) {}
    83→        }
    84→        if (!theme) { theme = window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light'; }
    85→        var splash = document.querySelector('.spash-waiting-for-js');
    86→        splash.setAttribute('data-theme', theme);
    87→        if (theme !== 'dark') { splash.style.backgroundColor = accent; splash.style.color = 'white'; }
    88→        splash.hidden = false;
    89→        var splashErrorTitle = document.querySelector('.splash-error-title');
    90→        var splashError = document.querySelector('.splash-errors');
    91→        var splashReset = document.querySelector('.splash-reset');

    93→        window.onerror = function (event, source, line, col, err) {
    94→          if (++totalErrors > 5) return;

    96→          var el = document.createElement('div');

    98→          splashError.appendChild(el);
    99→        };

   101→          splashReset.disabled = true;
   102→          function hardRefresh() {
   103→            var search = location.search.replace(/[?&]nocache=[\d.]+(?=$|&)/, '');
   104→            document.cookie = 'tw_clear_cache_once=1; max-age=60; path=/; samesite=strict; secure';
   105→            location.replace(location.pathname + search + (search ? '&' : '?') + 'nocache=' + Math.floor(Math.random() * 10000000));
   106→          }
   107→          if ('serviceWorker' in navigator) {
   108→            setTimeout(hardRefresh, 5000);
   109→            navigator.serviceWorker.getRegistration('')
   110→              .then(function(registration) { if (registration) { return registration.unregister(); } })
   111→              .then(hardRefresh).catch(hardRefresh);
   112→          } else { hardRefresh(); }
   113→        };
   114→        window.SplashEnd = () => { splash.hidden = true; window.onerror = null; };
